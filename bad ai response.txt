@Composable
fun ChatEmptyState(modifier: Modifier = Modifier) {
    val accentColor = LocalAccentColor.current
    
    EmptyStateContainer(
        title = "Personal, AI",
        subtitle = "I can try.",
        hint = "Ask me anything about your notes, or share what's on your mind.",
        modifier = modifier
    ) {
        val infiniteTransition = rememberInfiniteTransition(label = "chat_pulse")
        val phase by infiniteTransition.animateFloat(
            initialValue = 0f,
            targetValue = 2f * PI.toFloat(),
            animationSpec = infiniteRepeatable(tween(2000, easing = LinearEasing)),
            label = "phase"
        )
        
        Canvas(modifier = Modifier.size(120.dp)) {
            val width = size.width
            val height = size.height
            
            // Draw a subtle waveform
            val path = Path()
            val steps = 20
            val stepSize = width / steps
            
            path.moveTo(0f, height / 2)
            
            for (i in 0..steps) {
                val x = i * stepSize
                // Sine wave that breathes in amplitude
                val amplitude = (height * 0.2f) * sin(phase) * sin((i.toFloat() / steps) * PI.toFloat())
                val y = height / 2 + amplitude * sin(phase * 2 + i * 0.5f)
                path.lineTo(x, y.toFloat())
            }
            
            drawPath(
                path = path,
                color = accentColor.copy(alpha = 0.6f),
                style = Stroke(width = 3.dp.toPx(), cap = StrokeCap.Round)
            )
            
            // Central guiding dot
            drawCircle(
                color = accentColor,
                radius = 4.dp.toPx(),
                center = Offset(width / 2 + (width * 0.3f * sin(phase)), height / 2)
            )
        }
    }
}
