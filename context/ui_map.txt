**SSE Stream Handler:**
```kotlin
LaunchedEffect(sessionId) {
    viewModel.connectToStream(query, sessionId)
        .collect { event ->
            when (event) {
                is AgentEvent.Processing -> {
                    state = state.copy(
                        agentActivity = AgentActivity.Thinking(event.content)
                    )
                }
                is AgentEvent.ToolCall -> {
                    state = state.copy(
                        agentActivity = AgentActivity.UsingTool(
                            event.toolName,
                            event.status
                        )
                    )
                }
                is AgentEvent.Result -> {
                    appendToLastMessage(event.content)
                    if (event.isFinal) {
                        state = state.copy(isProcessing = false)
                    }
                }
                is AgentEvent.Command -> {
                    executeCommand(event.command)
                }
            }
        }
}
```

private val permissionLauncher = registerForActivityResult(

    _micPermissionGranted.value = permissions[RECORD_AUDIO] ?: false

## INTENT HANDLING

**Share Intent:**
```kotlin
private fun handleIntent(intent: Intent?) {
    when (intent?.action) {
        Intent.ACTION_SEND -> {
            val text = intent.getStringExtra(Intent.EXTRA_TEXT)
            viewModel.handleShareText(text)
        }
        Intent.ACTION_SEND_MULTIPLE -> {
            val uris = intent.getParcelableArrayListExtra<Uri>(Intent.EXTRA_STREAM)
            viewModel.handleShareFiles(uris)
        }
    }
}
```
