# STATE MANAGEMENT

## Overview

This document describes how state is managed across the Smarty application, including local state, server state, and synchronization mechanisms.

---

## ANDROID STATE MANAGEMENT

### Architecture Pattern

**Pattern:** MVVM with Unidirectional Data Flow

**Components:**
- **ViewModel:** State holder and business logic
- **StateFlow:** Reactive state streams
- **Compose:** Declarative UI based on state

### State Flow Hierarchy

```
ViewModel (State Holder)
    │
    ├─> StateFlow<State>
    │       │
    │       └─> collectAsState() in Composable
    │               │
    │               └─> UI renders based on state
    │
    └─> User Actions
            │
            └─> ViewModel functions update state
```

```kotlin
class SmartyViewModel(...) : ViewModel() {
    
    private val _notes = MutableStateFlow<List<Note>>(emptyList())
    val notes: StateFlow<List<Note>> = _notes.asStateFlow()
    
    private val _isProcessing = MutableStateFlow(false)
    val isProcessing: StateFlow<Boolean> = _isProcessing.asStateFlow()
    
    fun addNote(content: String, attachments: List<Attachment>) {
        viewModelScope.launch {
            _isProcessing.value = true
            try {
                val newNote = repository.addNote(content, attachments)
                _notes.value = _notes.value + newNote
            } finally {
                _isProcessing.value = false
            }
        }
    }
}
```

**SSE Stream Handling:**
```kotlin
fun connectToStream(query: String, sessionId: String) {
    viewModelScope.launch {
        _connectionState.value = ConnectionState.CONNECTING
        
        try {
            api.chatStream(query, sessionId)
                .collect { event ->
                    when (event) {
                        is AgentEvent.Processing -> {
                            _typingIndicator.value = true
                            _agentActivity.value = AgentActivity.Thinking(event.content)
                        }
                        is AgentEvent.ToolCall -> {
                            _agentActivity.value = AgentActivity.UsingTool(
                                event.toolName,
                                event.status
                            )
                        }
                        is AgentEvent.Result -> {
                            appendToMessage(event.content)
                            if (event.isFinal) {
                                _isSending.value = false
                                _typingIndicator.value = false
                            }
                        }
                        is AgentEvent.Command -> {
                            handleCommand(event.command)
                        }
                    }
                }
        } catch (e: Exception) {
            _connectionState.value = ConnectionState.ERROR
        }
    }
}
```

**Session Tracking:**

```kotlin
object ActiveSessionManager {
    private val activeSessions = ConcurrentHashMap<String, SessionInfo>()
    
    fun startSession(userId: String, sessionId: String, type: String) {
        activeSessions[sessionId] = SessionInfo(
            userId = userId,
            sessionId = sessionId,
            type = type,
            startedAt = System.currentTimeMillis()
        )
    }
    
    fun endSession(userId: String, sessionId: String) {
        activeSessions.remove(sessionId)
    }
    
    fun getActiveSession(userId: String): SessionInfo? {
        return activeSessions.values.find { it.userId == userId }
    }
}
```

### Thinking Storage

**File:** `server/src/main/kotlin/com/example/smarty/server/agent/ThinkingStorageManager.kt`

**Progressive Thinking Persistence:**
```kotlin
object ThinkingStorageManagerSingleton {
    private val thinkingCache = ConcurrentHashMap<String, StringBuilder>()
    
    fun appendThinking(sessionId: String, thinking: String) {
        thinkingCache
            .computeIfAbsent(sessionId) { StringBuilder() }
            .append(thinking)
    }
    
    fun getCurrentThinking(sessionId: String): String {
        return thinkingCache[sessionId]?.toString() ?: ""
    }
    
    fun finalizeAndGetThinking(sessionId: String): String {
        val thinking = getCurrentThinking(sessionId)
        thinkingCache.remove(sessionId)
        return thinking
    }
}
```

**File:** `app/src/main/java/com/example/smarty/data/worker/SyncWorker.kt`

**Sync Scheduling:**
```kotlin
class SyncWorker(...) : CoroutineWorker(...) {
    
    override suspend fun doWork(): Result {
        return try {
            // Push local changes
            val pendingOps = syncQueueDao.getPendingOperations()
            if (pendingOps.isNotEmpty()) {
                api.pushSync(pendingOps)
                syncQueueDao.markProcessed(pendingOps)
            }
            
            // Pull remote changes
            val lastSyncAt = syncStateDao.getLastSyncAt()
            val changes = api.pullSync(lastSyncAt)
            applyChanges(changes)
            
            Result.success()
        } catch (e: Exception) {
            Result.retry()
        }
    }
    
    companion object {
        fun schedule(context: Context) {
            val workRequest = PeriodicWorkRequestBuilder<SyncWorker>(15, TimeUnit.MINUTES)
                .build()
            WorkManager.getInstance(context).enqueue(workRequest)
        }
    }
}
```

## STATE RECOVERY

### Checkpoint System

**File:** `server/src/main/kotlin/com/example/smarty/server/agent/AgentPersistenceManager.kt`

**Checkpoint Storage:**
```kotlin
class AgentPersistenceManager(private val userId: String) {
    
    suspend fun saveCheckpoint(sessionId: String, state: AgentState) {
        val checkpoint = AgentCheckpoint(
            sessionId = sessionId,
            userId = userId,
            stateJson = Json.encodeToString(state),
            version = state.version
        )
        repository.saveCheckpoint(checkpoint)
    }
    
    suspend fun loadCheckpoint(sessionId: String): AgentState? {
        val checkpoint = repository.getLatestCheckpoint(sessionId)
        return checkpoint?.let {
            Json.decodeFromString(it.stateJson)
        }
    }
}
```

## STATE CLEANUP

### Session Expiration

**Server:**
```kotlin
// Expire sessions after 30 days
fun cleanupExpiredSessions() {
    val expiryThreshold = System.currentTimeMillis() - (30 * 24 * 60 * 60 * 1000)
    sessions.entries.removeIf { (_, session) ->
        session.lastInteractedAt < expiryThreshold
    }
}
```

### Cache Cleanup

**Android:**
```kotlin
class CacheCleanupWorker(...) : CoroutineWorker(...) {
    
    override suspend fun doWork(): Result {
        val cacheDir = applicationContext.cacheDir
        val threshold = System.currentTimeMillis() - (7 * 24 * 60 * 60 * 1000)
        
        cacheDir.listFiles()?.forEach { file ->
            if (file.lastModified() < threshold) {
                file.delete()
            }
        }
        
        return Result.success()
    }
}
```

## STATE DIAGRAMS

### Chat Message State

```
CREATED → SENDING → SENT → ACKNOWLEDGED
            │          │
            └─> FAILED ─┘
```

### Sync State

```
IDLE → SYNCING → SYNCED
       │          │
       └─> ERROR <─┘
```

### Agent Activity State

```
IDLE → THINKING → USING_TOOL → RESPONDING → IDLE
         │              │           │
         └──────────────┴───────────┘
                    (direct response)
```

### Note State

```
DRAFT → SAVED → SYNCED → ARCHIVED
                │          │
                └─> DELETED <─┘
```
