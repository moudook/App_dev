## TEST EXAMPLES

### ViewModel Test Example

```kotlin
class SmartyViewModelTest {

    private lateinit var viewModel: SmartyViewModel
    private lateinit var repository: MockNoteRepository
    private lateinit var testDispatcher: TestDispatcher

    @Before
    fun setup() {
        testDispatcher = StandardTestDispatcher()
        repository = MockNoteRepository()
        viewModel = SmartyViewModel(repository, Dispatchers.Main = testDispatcher)
    }

    @Test
    fun `addNote should add note to list`() = runTest {
        // Given
        val noteContent = "Test note"
        
        // When
        viewModel.addNote(noteContent, emptyList())
        testDispatcher.scheduler.advanceUntilIdle()
        
        // Then
        val notes = viewModel.notes.value
        assertEquals(1, notes.size)
        assertEquals(noteContent, notes.first().content)
    }

    @Test
    fun `deleteNote should remove note from list`() = runTest {
        // Given
        viewModel.addNote("Note 1", emptyList())
        viewModel.addNote("Note 2", emptyList())
        testDispatcher.scheduler.advanceUntilIdle()
        
        val noteToDelete = viewModel.notes.value.first()
        
        // When
        viewModel.deleteNote(noteToDelete)
        testDispatcher.scheduler.advanceUntilIdle()
        
        // Then
        assertEquals(1, viewModel.notes.value.size)
    }
}
```

### Agent Test Example

```kotlin
class ServerAgentTest {

    private lateinit var agent: ServerAgent
    private lateinit var llmProvider: MockLlmProvider
    private lateinit var tavilyTool: MockTavilySearchTool
    private lateinit var vectorStore: MockVectorStore

    @Before
    fun setup() {
        llmProvider = MockLlmProvider()
        tavilyTool = MockTavilySearchTool()
        vectorStore = MockVectorStore()
        
        agent = ServerAgent(
            llmProvider = llmProvider,
            tavilyTool = tavilyTool,
            vectorStore = vectorStore,
            eventEmitter = { /* mock emitter */ },
            userId = "test-user"
        )
    }

    @Test
    fun `run should emit processing events`() = runTest {
        // Given
        llmProvider.responseChunks = listOf(
            LlmChunk(content = "Thinking"),
            LlmChunk(content = " about"),
            LlmChunk(content = " this")
        )
        
        val events = mutableListOf<AgentEvent>()
        agent = agent.copy(eventEmitter = { events.add(it) })
        
        // When
        agent.run("Test query")
        
        // Then
        assertTrue(events.any { it is AgentEvent.Processing })
    }

    @Test
    fun `run should call search tool for research queries`() = runTest {
        // Given
        val query = "Research climate change"
        
        // When
        agent.run(query)
        
        // Then
        assertTrue(tavilyTool.searchCalled)
    }
}
```

### Route Test Example

```kotlin
class ChatRoutesTest {

    private lateinit var application: Application
    private lateinit var testClient: TestApplicationEngine

    @Before
    fun setup() {
        testClient = TestApplicationEngine {
            module()
        }
        testClient.start()
    }

    @Test
    fun `chat stream should require authentication`() = runTest {
        // When
        val response = testClient.handleRequest {
            addRequestHeader("Authorization", "")
            get("/chat/stream?query=test")
        }

        // Then
        assertEquals(HttpStatusCode.Unauthorized, response.response.status())
    }

    @Test
    fun `chat stream should return SSE events`() = runTest {
        // Given
        val token = createTestToken()
        
        // When
        val response = testClient.handleRequest {
            addRequestHeader("Authorization", "Bearer $token")
            get("/chat/stream?query=test")
        }

        // Then
        assertEquals(HttpStatusCode.OK, response.response.status())
        assertEquals("text/event-stream", response.response.headers["Content-Type"])
    }
}
```

## CI/CD INTEGRATION

### Recommended Pipeline

```yaml
# .github/workflows/test.yml
name: Tests

on: [push, pull_request]

jobs:
  test-android:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Set up JDK
        uses: actions/setup-java@v3
        with:
          java-version: '17'
      - name: Run Android tests
        run: ./gradlew :app:test
      - name: Upload coverage
        uses: codecov/codecov-action@v3

  test-server:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Set up JDK
        uses: actions/setup-java@v3
        with:
          java-version: '17'
      - name: Run server tests
        run: ./gradlew :server:test
      - name: Upload coverage
        uses: codecov/codecov-action@v3

  integration-tests:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: ankane/pgvector

          POSTGRES_DB: test_db

          POSTGRES_PASSWORD: test_pass
        ports:
          - 5432:5432
    steps:
      - uses: actions/checkout@v3
      - name: Run integration tests
```
